// led_blink.ino (v0.1) - first version, blocking design
int led = 13;
int button = 2;
int blinking = 1;

void setup() {
  pinMode(led, OUTPUT);
  pinMode(button, INPUT);
}

void loop() {
  if (digitalRead(button) == HIGH) {
    blinking = !blinking;
  }
  if (blinking) {
    digitalWrite(led, HIGH);
    delay(500);
    digitalWrite(led, LOW);
    delay(500);
  }
}
